# ECHOFIELD FM-1 on the host simulator: the bank, the session, the panel, presets, RESAMPLE, the USB commands.
#   tools/build.py app --host, then app/build/host/echofield_host app/tests/basic.txt OUTDIR
# Parameter ids (dsp/ef_dsp.h): 1 TEXTURE, 2 SCATTER, 5 PITCH, 49 MODE.

# ---- an empty FM-1: the seed plays, nothing else is there
wait 1200
expect slots_ok 0
expect slot -1
expect play 0
expect free 92
peakreset
wait 1500
expect peak_db_max -80
tap PLAY
expect play 1
wait 2500
peakreset
wait 1000
expect peak_db_min -30
expect peak_db_max 0

# ---- samples into the bank: the seed gives way, the sectors are counted
bankwrite 0 3 ALPHA
expect slots_ok 1
expect slot 0
expect free 83
bankwrite 1 2 BETA
expect slots_ok 2
expect free 77
expect slot 0
press EDIT
wait 40
key w1 down
wait 40
key w1 up
wait 40
release EDIT
wait 300
expect slot 1
peakreset
wait 1500
expect peak_db_min -30

# ---- the session: saved once the sound has stopped, back after a reboot
tap PLAY
expect play 0
wait 6000
expect dirty 0
reboot
wait 1200
expect slots_ok 2
expect slot 1
expect play 0

# ---- a damaged sample is found at boot and never played
bankbreak 1
reboot
wait 1200
expect state1 2
expect slots_ok 1
expect slot 0
bankerase 1
expect state1 0
expect free 83

# ---- knobs, locks, RANDOM, undo
expect page 0

expect pq1 50462

expect pq1 50462
press ARP
turn K1 1
release ARP
expect locked1 0
press OCT-
tap OCT+
release OCT-
expect pq1 50462
turn ALGO 1

turn ALGO -1
expect mode 0
